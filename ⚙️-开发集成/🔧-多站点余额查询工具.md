## 功能介绍

### 余额查询

* **单个查询**：点击卡片上的"🔄 刷新"按钮- **批量查询**：点击顶部"🔄 查询所有余额"按钮- **自动刷新**：系统每分钟自动更新一次余额

### 余额显示

* 主显示：人民币金额（¥）- 副显示：美元金额（$）- 总计卡片：显示所有站点余额总和

### 汇率设置每个站点可独立设置汇率：

* 展开站点卡片- 找到"汇率"输入框- 输入新汇率（默认1美元=7人民币）- 自动保存并更新显示

### 警戒线功能

* 设置低余额警戒值（单位：人民币）- 余额低于警戒线时显示红色警告- 触发声音报警提醒

### 历史记录

* 点击"📈 历史"查看36小时内余额变化- 图表展示余额趋势- 总计卡片也有独立的历史记录

### 站点管理

* **编辑信息**：点击站点名称、域名、API用户等字段直接编辑- **充值**：点击"💵 充值"跳转到站点充值页面- **删除**：点击"🗑️ 删除"移除站点

### 数据管理

* **导出配置**：点击"📥 导出配置"保存站点信息- **导入配置**：点击"📤 导入配置"恢复站点信息- **本地存储**：所有数据自动保存在浏览器本地

## 常见问题

### Q: 查询失败提示CORS错误？**A:** 确保CORS Unblock插件已启用（图标为绿色）。

### Q: 余额显示不正确？**A:** 检查API配置是否正确，特别是Authorization token格式。

### Q: 历史记录丢失？**A:** 历史数据保存在浏览器本地，清除浏览器数据会导致丢失。建议定期导出配置备份。

### Q: 报警声音太吵？**A:** 点击报警弹窗的"关闭报警"按钮，或调高警戒线数值。

### Q: 如何批量添加站点？**A:** 可以先添加一个站点，导出配置文件，编辑JSON文件批量添加，再导入。

## 安全提示

* **Token安全**：请勿将包含Token的配置文件分享给他人- **本地使用**：建议仅在个人电脑上使用，避免在公共设备使用- **定期备份**：重要数据请定期导出备份

## 技术支持如遇到问题，请检查：

* CORS插件是否正常启用- 网络连接是否正常- API接口是否可访问- Token格式是否正确

// 创建报警声音（使用Web Audio API生成）
function createAlarmSound() {
    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
    
    return {
        start: () => {
            if (!alarmActive) {
                alarmActive = true;
                playBeep();
            }
            
            function playBeep() {
                if (!alarmActive) return;
                
                const oscillator = audioContext.createOscillator();
                const gainNode = audioContext.createGain();
                
                oscillator.connect(gainNode);
                gainNode.connect(audioContext.destination);
                
                oscillator.type = 'sine';
                oscillator.frequency.setValueAtTime(880, audioContext.currentTime);
                oscillator.frequency.exponentialRampToValueAtTime(440, audioContext.currentTime + 0.5);
                
                gainNode.gain.setValueAtTime(0, audioContext.currentTime);
                gainNode.gain.linearRampToValueAtTime(0.3, audioContext.currentTime + 0.05);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.5);
                
                oscillator.start(audioContext.currentTime);
                oscillator.stop(audioContext.currentTime + 0.5);
                
                // 循环播放
                setTimeout(() => {
                    if (alarmActive) playBeep();
                }, 1000);
            }
        },
        stop: () => {
            alarmActive = false;
        }
    };
}

// 加载站点数据
function loadSites() {
    const savedSites = localStorage.getItem('balanceCheckerSites');
    if (savedSites) {
        sites = JSON.parse(savedSites);
    }
    
    // 加载历史记录
    const savedHistory = localStorage.getItem('balanceHistory');
    if (savedHistory) {
        balanceHistory = JSON.parse(savedHistory);
        cleanOldHistory(); // 清理超过36小时的数据
    }
}

// 清理超过36小时的历史数据
function cleanOldHistory() {
    const now = Date.now();
    const maxAge = 36 * 60 * 60 * 1000; // 36小时
    
    for (const siteId in balanceHistory) {
        if (balanceHistory[siteId]) {
            balanceHistory[siteId] = balanceHistory[siteId].filter(
                record => (now - record.timestamp)  200) {
        balanceHistory[siteId].shift();
    }
    
    localStorage.setItem('balanceHistory', JSON.stringify(balanceHistory));
}

function editField(element) {
    // 如果已经在编辑中，返回
    if (element.querySelector('input')) return;
    
    const siteId = element.dataset.site;
    const field = element.dataset.field;
    const site = sites.find(s => s.id === siteId);
    if (!site) return;
    
    // 获取当前值
    let currentValue = '';
    if (field === 'url') {
        currentValue = site.url.replace(/\/$/, '');
    } else if (field === 'apiUser') {
        currentValue = site.apiUser;
    } else if (field === 'authorization') {
        currentValue = site.authorization;
    }
    
    // 创建输入框
    const input = document.createElement('input');
    input.type = field === 'authorization' ? 'password' : 'text';
    input.value = currentValue;
    input.style.cssText = `
        width: 100%;
        padding: 2px 4px;
        font-size: 12px;
        border: 1px solid #3b82f6;
        border-radius: 4px;
        background: white;
        color: #1f2937;
    `;
    
    // 保存原始内容
    const originalContent = element.innerHTML;
    element.innerHTML = '';
    element.appendChild(input);
    
    // 聚焦并选中
    input.focus();
    input.select();
    
    // 保存更改
    function saveEdit() {
        const newValue = input.value.trim();
        if (newValue) {
            if (field === 'url') {
                // 处理URL
                let normalizedUrl = newValue;
                if (!normalizedUrl.startsWith('http://') && !normalizedUrl.startsWith('https://')) {
                    normalizedUrl = 'https://' + normalizedUrl;
                }
                normalizedUrl = normalizedUrl.endsWith('/') ? normalizedUrl : normalizedUrl + '/';
                site.url = normalizedUrl;
            } else if (field === 'apiUser') {
                site.apiUser = newValue;
            } else if (field === 'authorization') {
                site.authorization = newValue.startsWith('Bearer ') ? newValue : `Bearer ${newValue}`;
            }
            
            saveSites();
            renderSites();
            showToast('信息已更新');
        } else {
            // 恢复原始内容
            element.innerHTML = originalContent;
        }
    }
    
    // 取消编辑
    function cancelEdit() {
        element.innerHTML = originalContent;
    }
    
    // 监听事件
    input.addEventListener('blur', saveEdit);
    input.addEventListener('keydown', (e) => {
        e.stopPropagation();
        if (e.key === 'Enter') {
            saveEdit();
        } else if (e.key === 'Escape') {
            cancelEdit();
        }
    });
    
    // 阻止点击事件冒泡
    input.onclick = (e) => e.stopPropagation();
}

// 查询单个站点余额
async function querySingleBalance(siteId) {
    const site = sites.find(s => s.id === siteId);
    if (!site) return;
    
    const balanceElement = document.getElementById(`balance-${siteId}`);
    balanceElement.className = 'site-balance loading';
    balanceElement.innerHTML = ' 查询中...';
    
    try {
        const response = await fetch(`${site.url}api/user/self`, {
            method: 'GET',
            headers: {
                'Rix-Api-User': site.apiUser,
                'Authorization': site.authorization
            },
            mode: 'cors'
        });
        
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        
        const data = await response.json();
        
        // 从API返回的数据中提取余额
        const quotaValue = data.data?.quota || data.quota || 0;
        const balance = (quotaValue / 500000).toFixed(2);
        
        site.balance = balance;
        site.quotaValue = quotaValue;
        site.lastQuery = new Date().toISOString();
        saveSites();
        
        // 记录历史
        recordBalanceHistory(siteId, balance);
        
        balanceElement.className = 'site-balance';
        balanceElement.textContent = `$${balance}`;
        
        // 更新详细信息
        renderSites();
        
        // 更新总余额历史
        updateTotalHistory();
        
        // 检查是否需要报警
        checkAlarms();
        
    } catch (error) {
        console.error('查询失败:', error);
        balanceElement.className = 'site-balance error';
        
        if (error.message.includes('Failed to fetch')) {
            balanceElement.textContent = 'CORS错误';
        } else {
            balanceElement.textContent = '查询失败';
        }
    }
}

function updateTotalHistory() {
    let totalBalanceCNY = 0;
    let totalBalanceUSD = 0;
    let hasBalance = false;
    
    sites.forEach(site => {
        if (site.balance !== null) {
            hasBalance = true;
            totalBalanceUSD += parseFloat(site.balance);
            totalBalanceCNY += parseFloat(site.balance) * (site.exchangeRate || 7);
        }
    });
    
    if (hasBalance) {
        // 初始化总计历史
        if (!balanceHistory['total']) {
            balanceHistory['total'] = [];
        }
        
        // 添加新记录
        balanceHistory['total'].push({
            timestamp: Date.now(),
            totalUSD: totalBalanceUSD,
            totalCNY: totalBalanceCNY
        });
        
        // 只保留36小时内的记录
        const cutoffTime = Date.now() - (36 * 60 * 60 * 1000);
        balanceHistory['total'] = balanceHistory['total'].filter(h => h.timestamp > cutoffTime);
        
        // 保存历史
        saveBalanceHistory();
    }
}

function importData(event) {
    const file = event.target.files[0];
    if (!file) return;
    
    const reader = new FileReader();
    reader.onload = (e) => {
        try {
            const importedSites = JSON.parse(e.target.result);
            
            if (!Array.isArray(importedSites)) {
                throw new Error('无效的配置文件');
            }
            
            // 合并或替换现有配置
            if (confirm('是否替换现有配置？选择"取消"将合并配置。')) {
                sites = importedSites;
            } else {
                // 合并配置，避免重复
                importedSites.forEach(importedSite => {
                    if (!sites.find(s => s.url === importedSite.url && s.apiUser === importedSite.apiUser)) {
                        importedSite.id = Date.now().toString() + Math.random();
                        sites.push(importedSite);
                    }
                });
            }
            
            saveSites();
            renderSites();
            showToast('配置导入成功');
            
        } catch (error) {
            showToast('导入失败: ' + error.message, 'error');
        }
    };
    
    reader.readAsText(file);
    event.target.value = ''; // 清空input，允许重复导入同一文件
}
